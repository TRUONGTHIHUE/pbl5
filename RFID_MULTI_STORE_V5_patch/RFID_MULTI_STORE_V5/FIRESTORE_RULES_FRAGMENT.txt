// Keep stores/{storeId} readable only by its authenticated owner.
match /stores/{storeId}/authorizedDevices/{deviceId} {
  function isOwner() {
    return request.auth != null &&
      get(/databases/$(database)/documents/stores/$(storeId)).data.ownerUid == request.auth.uid &&
      deviceId.matches('^[A-Za-z0-9_-]{3,64}$');
  }
  allow get, list: if isOwner();
  allow create, update: if isOwner() &&
    request.resource.data.keys().hasOnly(['enabled']) &&
    request.resource.data.enabled is bool;
  allow delete: if isOwner();
}
